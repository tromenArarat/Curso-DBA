CREATE TABLE ALMACEN (
    NRO INT AUTO_INCREMENT,
    NOMBRE VARCHAR(50),
    RESPONSABLE VARCHAR(50),
    PRIMARY KEY (NRO)
);

CREATE TABLE ARTICULO (
    COD_ART INT AUTO_INCREMENT,
    DESCRIPCION VARCHAR(4000),
    PRECIO DECIMAL(10,2),
    PRIMARY KEY (COD_ART)
);

CREATE TABLE MATERIAL (
    COD_MAT INT AUTO_INCREMENT,
    DESCRIPCION VARCHAR(4000),
    PRIMARY KEY (COD_MAT)
);

CREATE TABLE CIUDAD (
    COD_CIU INT AUTO_INCREMENT,
    NOMBRE VARCHAR(50),
    PRIMARY KEY (COD_CIU)
);

CREATE TABLE PROVEEDOR (
    COD_PROV INT AUTO_INCREMENT,
    NOMBRE VARCHAR(50),
    DOMICILIO VARCHAR(255),
    COD_CIU INT,
    FECHA_ALTA DATETIME,
    PRIMARY KEY (COD_PROV),
    CONSTRAINT fk_proveedor_ciudad FOREIGN KEY (COD_CIU) REFERENCES CIUDAD(COD_CIU)
);

CREATE TABLE CONTIENE (
    NRO INT,
    COD_ART INT,
    PRIMARY KEY (NRO, COD_ART),
    CONSTRAINT fk_contene_almacen FOREIGN KEY (NRO) REFERENCES ALMACEN(NRO),
    CONSTRAINT fk_contene_articulo FOREIGN KEY (COD_ART) REFERENCES ARTICULO(COD_ART)
);

CREATE TABLE COMPUESTO_POR (
    COD_ART INT, 
    COD_MAT INT,
    PRIMARY KEY (COD_ART, COD_MAT),
    CONSTRAINT fk_compuesto_articulo FOREIGN KEY (COD_ART) REFERENCES ARTICULO(COD_ART),
    CONSTRAINT fk_compuesto_material FOREIGN KEY (COD_MAT) REFERENCES MATERIAL(COD_MAT)
);
   
CREATE TABLE PROVISTO_POR (
    COD_MAT INT,
    COD_PROV INT,
    PRIMARY KEY (COD_MAT, COD_PROV),
    CONSTRAINT fk_provisto_material FOREIGN KEY (COD_MAT) REFERENCES MATERIAL(COD_MAT),
    CONSTRAINT fk_provisto_proveedor FOREIGN KEY (COD_PROV) REFERENCES PROVEEDOR(COD_PROV)
);
